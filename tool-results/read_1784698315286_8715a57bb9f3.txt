   200→   200→function applyEffects(state: GameState, effects: CardEffects): Partial<GameState> {
   201→   201→  const next: Partial<GameState> = {};
   202→   202→  if (effects.nuclearProgress) next.nuclearProgress = clamp(state.nuclearProgress + effects.nuclearProgress);
   203→   203→  if (effects.regionalInfluence) next.regionalInfluence = clamp(state.regionalInfluence + effects.regionalInfluence);
   204→   204→  if (effects.economicStability) next.economicStability = clamp(state.economicStability + effects.economicStability);
   205→   205→  if (effects.domesticSupport) next.domesticSupport = clamp(state.domesticSupport + effects.domesticSupport);
   206→   206→  if (effects.militaryCapability) next.militaryCapability = clamp(state.militaryCapability + effects.militaryCapability);
   207→   207→  if (effects.deterrence) next.deterrence = clamp(state.deterrence + effects.deterrence);
   208→   208→  if (effects.warEscalation) {
   209→   209→    const delta = (effects.warEscalation - 1) * 25;
   210→   210→    next.warEscalation = clamp(state.warEscalation + delta);
   211→   211→  }
   212→   212→  if (effects.nuclearBreakout) next.nuclearBreakoutMult = state.nuclearBreakoutMult * effects.nuclearBreakout;
   213→   213→  if (effects.regimeChange) next.regimeChangeMult = state.regimeChangeMult * effects.regimeChange;
   214→   214→  if (effects.negotiationChance) next.negotiationChanceMult = state.negotiationChanceMult * effects.negotiationChance;
   215→   215→  if (effects.usWithdrawal) next.usWithdrawalMult = state.usWithdrawalMult * effects.usWithdrawal;
   216→   216→  if (effects.israelIsolation) next.israelIsolationMult = state.israelIsolationMult * effects.israelIsolation;
   217→   217→  return next;
   218→   218→}

   223→   223→function selectEnemyPlay(state: GameState): GameCard | null {
   224→   224→  const allEnemies = [...usCards, ...israelCards, ...arabCards, ...natoCards];
   225→   225→  const available = allEnemies.filter(card => {
   226→   226→    const cost = getCardCost(card.id);
   227→   227→    if (cost > state.enemyElixir) return false;
   228→   228→    if (state.enemyPlays.find(p => p.card.id === card.id)) return false;
   229→   229→    if (card.requiresHighWarEscalation && state.warEscalation < 75) return false;
   230→   230→    return true;
   231→   231→  });
   232→   232→  if (available.length === 0) return null;
   233→   233→
   234→   234→  // Weighted random based on situation
   235→   235→  const weighted = available.map(card => {
   236→   236→    let weight = card.aiWeight ?? 10;
   237→   237→    // Escalate if war high
   238→   238→    if (state.warEscalation > 70 && card.category === "military") weight *= 1.5;
   239→   239→    if (state.warEscalation > 70 && card.rarity === "apocalyptic") weight *= 1.8;
   240→   240→    // React to Iran nuclear progress
   241→   241→    if (state.nuclearProgress > 75 && (card.id === "israel_preemptive" || card.id === "us_strike_nukes")) weight *= 2.5;
   242→   242→    if (state.nuclearProgress > 60 && card.id === "israel_nuclear_facility") weight *= 2;
   243→   243→    // React to Iran Hormuz
   244→   244→    if (state.playedIranCardIds.includes("iran_hormuz")) {
   245→   245→      if (card.id === "us_hormuz_operation" || card.id === "us_oil_blockade") weight *= 3;
   246→   246→    }
   247→   247→    // React to Iran proxy activity
   248→   248→    if (state.playedIranCardIds.some(id => ["iran_houthi", "iran_hezbollah_full", "iran_iraq_militias"].includes(id))) {
   249→   249→      if (card.id === "us_strike_iraq_militias" || card.id === "israel_hezbollah_war") weight *= 2;
   250→   250→    }
   251→   251→    // Prefer cheaper cards early, expensive later
   252→   252→    if (state.turn <= 3 && getCardCost(card.id) <= 4) weight *= 1.3;
   253→   253→    if (state.turn >= 5 && getCardCost(card.id) >= 5) weight *= 1.4;
   254→   254→    return { card, weight };
   255→   255→  });
   256→   256→
   257→   257→  const total = weighted.reduce((s, w) => s + w.weight, 0);
   258→   258→  let r = Math.random() * total;
   259→   259→  for (const item of weighted) {
   260→   260→    r -= item.weight;
   261→   261→    if (r <= 0) return item.card;
   262→   262→  }
   263→   263→  return weighted[weighted.length - 1].card;
   264→   264→}

   269→   269→function spawnArenaUnit(card: GameCard, side: "iran" | "enemy"): ArenaUnit {
   270→   270→  // Different card types have different arena behaviors
   271→   271→  let hp = 50;
   272→   272→  let attack = 8;
   273→   273→  let speed = 6;
   274→   274→
   275→   275→  switch (card.category) {
   276→   276→    case "nuclear":
   277→   277→      hp = 20; attack = 100; speed = 12;
   278→   278→      break;
   279→   279→    case "military":
   280→   280→      hp = 70; attack = 12; speed = 8;
   281→   281→      break;
   282→   282→    case "proxy":
   283→   283→      hp = 40; attack = 6; speed = 7;
   284→   284→      break;
   285→   285→    case "cyber":
   286→   286→      hp = 25; attack = 15; speed = 15;
   287→   287→      break;
   288→   288→    case "diplomatic":
   289→   289→      hp = 35; attack = 4; speed = 4;
   290→   290→      break;
   291→   291→    case "economic":
   292→   292→      hp = 45; attack = 5; speed = 5;
   293→   293→      break;
   294→   294→    case "asymmetric":
   295→   295→      hp = 30; attack = 10; speed = 10;
   296→   296→      break;
   297→   297→    case "extreme":
   298→   298→      hp = 15; attack = 200; speed = 14;
   299→   299→      break;
   300→   300→    case "alliance":
   301→   301→      hp = 55; attack = 7; speed = 6;
   302→   302→      break;
   303→   303→    default:
   304→   304→      hp = 40; attack = 8; speed = 7;
   305→   305→  }
   306→   306→  // Rarity bonus
   307→   307→  if (card.rarity === "epic") { hp *= 1.2; attack *= 1.2; }
   308→   308→  if (card.rarity === "legendary") { hp *= 1.4; attack *= 1.4; }
   309→   309→  if (card.rarity === "apocalyptic") { hp *= 0.8; attack *= 2.5; speed *= 1.5; }
   310→   310→
   311→   311→  return {
   312→   312→    id: `${card.id}_${side}_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,
   313→   313→    cardId: card.id,
   314→   314→    card,
   315→   315→    side,
   316→   316→    position: side === "iran" ? 5 : 95,
   317→   317→    hp,
   318→   318→    attack,
   319→   319→    speed,
   320→   320→    state: "advancing",
   321→   321→    impactEffects: { target: side === "iran" ? "enemy" : "iran", effects: card.effects },
   322→   322→  };
   323→   323→}

   328→   328→function applyComboBonuses(state: GameState, baseEffects: CardEffects, playedCardIds: string[]): CardEffects {
   329→   329→  const activeCombos = detectActiveCombos(playedCardIds);
   330→   330→  if (activeCombos.length === 0) return baseEffects;
   331→   331→
   332→   332→  const boosted = { ...baseEffects };
   333→   333→  for (const combo of activeCombos) {
   334→   334→    if (combo.type === "axis_of_resistance") {
   335→   335→      boosted.regionalInfluence = (boosted.regionalInfluence || 0) * combo.multiplier;
   336→   336→    }
   337→   337→    if (combo.type === "nuclear_deterrence") {
   338→   338→      boosted.deterrence = (boosted.deterrence || 0) * combo.multiplier;
   339→   339→      boosted.nuclearBreakout = (boosted.nuclearBreakout || 1) * combo.multiplier;
   340→   340→    }
   341→   341→    if (combo.type === "asymmetric_full") {
   342→   342→      boosted.warEscalation = boosted.warEscalation || 1;
   343→   343→      boosted.economicStability = (boosted.economicStability || 0) * combo.multiplier;
   344→   344→    }
   345→   345→    if (combo.type === "eastern_axis") {
   346→   346→      boosted.economicStability = (boosted.economicStability || 0) * combo.multiplier;
   347→   347→    }
   348→   348→  }
   349→   349→  return boosted;
   350→   350→}

   355→   355→function selectEnemyResponses(state: GameState, iranCard: GameCard): GameCard[] {
   356→   356→  const allEnemies = [...usCards, ...israelCards, ...arabCards, ...natoCards];
   357→   357→  const weighted = allEnemies.map((card) => {
   358→   358→    let weight = card.aiWeight ?? 10;
   359→   359→    if (card.counters?.includes(iranCard.id)) weight *= 3.5;
   360→   360→    if (iranCard.id === "iran_nuclear_breakout") {
   361→   361→      if (card.id === "us_strike_nukes" || card.id === "israel_nuclear_facility" || card.id === "israel_preemptive") weight *= 3;
   362→   362→      if (card.id === "us_nuclear_umbrella" || card.id === "arab_saudi_nuke") weight *= 2.5;
   363→   363→    }
   364→   364→    if (iranCard.id === "iran_hormuz") {
   365→   365→      if (card.id === "us_hormuz_operation") weight *= 4;
   366→   366→      if (card.id === "us_oil_blockade") weight *= 2.5;
   367→   367→    }
   368→   368→    if (iranCard.id === "iran_hezbollah_full" || iranCard.id === "iran_hamas") {
   369→   369→      if (card.id === "israel_hezbollah_war" || card.id === "israel_hamas_war") weight *= 3;
   370→   370→      if (card.id === "us_aid_israel") weight *= 2.5;
   371→   371→    }
   372→   372→    if (iranCard.id === "iran_missile_strike") {
   373→   373→      if (card.id === "us_aid_israel" || card.id === "israel_air_strike") weight *= 2.5;
   374→   374→    }
   375→   375→    if (iranCard.id === "iran_diplomacy" && card.id === "us_negotiation_deception") weight *= 3;
   376→   376→    if (card.requiresHighWarEscalation && state.warEscalation < 75) weight *= 0.2;
   377→   377→    // Add counter system: check counteredBy
   378→   378→    const cardEnr = getCardEnrichment(card.id);
   379→   379→    if (cardEnr?.counteredBy?.includes(iranCard.id)) weight *= 0.3; // this card is countered by Iran card
   380→   380→    return { card, weight };
   381→   381→  });
   382→   382→
   383→   383→  const selected: GameCard[] = [];
   384→   384→  const actorCount: Record<string, number> = {};
   385→   385→  const targetCount = Math.random() < 0.4 ? 3 : 2;
   386→   386→
   387→   387→  for (let i = 0; i < targetCount; i++) {
   388→   388→    const available = weighted.filter((w) => {
   389→   389→      if (selected.find((s) => s.id === w.card.id)) return false;
   390→   390→      if (actorCount[w.card.actor] >= 2) return false;
   391→   391→      return true;
   392→   392→    });
   393→   393→    if (available.length === 0) break;
   394→   394→    const total = available.reduce((s, i) => s + i.weight, 0);
   395→   395→    let r = Math.random() * total;
   396→   396→    let chosen = available[available.length - 1].card;
   397→   397→    for (const item of available) {
   398→   398→      r -= item.weight;
   399→   399→      if (r <= 0) { chosen = item.card; break; }
   400→   400→    }
   401→   401→    selected.push(chosen);
   402→   402→    actorCount[chosen.actor] = (actorCount[chosen.actor] || 0) + 1;
   403→   403→  }
   404→   404→
   405→   405→  if ((iranCard.category === "alliance" || iranCard.id === "iran_diplomacy") && selected.length < 3) {
   406→   406→    const thirdParty = [...nkRussiaChinaCards];
   407→   407→    const tp = thirdParty[Math.floor(Math.random() * thirdParty.length)];
   408→   408→    if (!selected.find((s) => s.id === tp.id)) selected.push(tp);
   409→   409→  }
   410→   410→  return selected;
   411→   411→}

   413→   413→function checkEarlyEnding(state: GameState, iranCard: GameCard, enemyCards: GameCard[]): string | null {
   414→   414→  if (iranCard.id === "iran_nuclear_breakout" && state.nuclearProgress >= 85) return "iran_nuclear_deterrence";
   415→   415→  const allCards = [iranCard, ...enemyCards];
   416→   416→  if (allCards.some(c => c.id === "us_nuclear_strike" || c.id === "israel_nuclear_strike")) return "nuclear_war_regional";
   417→   417→  if (state.militaryCapability <= 15 && state.warEscalation >= 90) return "iran_strategic_defeat";
   418→   418→  if (state.economicStability <= 10 && state.domesticSupport <= 15) return "regime_change_from_within";
   419→   419→  if (state.deterrence >= 85 && state.regionalInfluence >= 85 && state.warEscalation < 40) return "us_withdrawal_ambition";
   420→   420→  if (state.negotiationChanceMult >= 2.5 && state.warEscalation < 30) return "comprehensive_peace";
   421→   421→  if (state.regionalInfluence >= 90 && state.israelIsolationMult >= 2.0) return "israel_strategic_weakening";
   422→   422→  return null;
   423→   423→}

   425→   425→function calculateStatChanges(before: GameState, after: GameState): Partial<Record<string, number>> {
   426→   426→  const changes: Partial<Record<string, number>> = {};
   427→   427→  const keys: (keyof GameState)[] = ["nuclearProgress", "regionalInfluence", "economicStability", "domesticSupport", "militaryCapability", "deterrence", "warEscalation"];
   428→   428→  for (const key of keys) {
   429→   429→    const diff = (after[key] as number) - (before[key] as number);
   430→   430→    if (diff !== 0) changes[key] = diff;
   431→   431→  }
   432→   432→  return changes;
   433→   433→}
